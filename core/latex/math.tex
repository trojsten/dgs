% Differential, optionally with a power and/or a subscript, written where the maths writes them:
%
%     \Diff{x}                      d x
%     \Diff^2{x}                    d^2 x
%     \FDiff_{\mathrm{vap}}{H}      \Delta_vap H      -- an enthalpy of vaporisation
%     \FDiff^2_{\mathrm{vap}}{H}    \Delta^2_vap H    -- both, in either order
%
% `e{^_}` is xparse's embellishment argument: it collects a `^{…}` and a `_{…}` in whichever
% order they appear, and neither has to be given. It replaces the older `[power]` optional, which
% nothing in the repository ever used and which was quietly dangerous: an `o` argument scans
% forward for `[`, so `\FDiff [A]` -- a concentration, an interval, a bracketed unit -- was
% swallowed as a superscript and set as \Delta^A. Embellishments never look for a bracket.
% A `[` straight after one of these is the retired `[power]` syntax. `t[` only reports whether the
% bracket is there, so the build stops with a message instead of setting something wrong: with no
% guard, `\FDiff[2]{x}` quietly sets as \Delta[2]x, and a silent wrong answer is the one outcome
% worth ruling out. (xparse's `!` prefix would let a *spaced* bracket through, which is what we
% would rather have, but this kernel -- LaTeX2e 2024-11-01 -- does not implement it.)
\NewDocumentCommand{\DiffBracket@}{m}{%
    \ClassError{dgs}{%
        \protect#1 does not take a bracketed argument%
    }{%
        The [power] form was retired when the differentials took embellishments.^^J%
        Write the power as a caret group and a subscript as an underscore group,^^J%
        for example \protect#1 caret 2 braces x.^^J%
        If you meant a bracketed quantity, put braces around it.%
    }%
}

\NewDocumentCommand{\Diff}{e{^_} t[ m}{%

    \diff@\IfValueT{#1}{^{#1}}\IfValueT{#2}{_{#2}} #4%
}
\NewDocumentCommand{\PDiff}{e{^_} t[ m}{%
    \IfBooleanT{#3}{\DiffBracket@{\PDiff}}%

% These nine are the whole set, and not for want of trying: halves, thirds, quarters and eighths
% are the only vulgar fractions MinionPro has a glyph for. Fifths, sixths, sevenths, ninths and
% tenths -- U+2150, U+2151, U+2152 and U+2155 to U+215A -- are absent from the font, and a missing
% glyph is not a compile error. xelatex writes `Missing character:` into the log and sets nothing
% at all, so a \TwoFifths would be a fraction that silently vanishes off the page. Hence there is
% no \TwoFifths: anything the font cannot set is written \frac, which is built rather than looked
% up in a glyph table, so it always exists.
%
% These nine are for a standalone value, above all a mixed number -- 33\OneThird, 666\TwoThirds.
% A coefficient inside a formula stays \frac: \frac{1}{2} m v^2, not \OneHalf m v^2. See CLAUDE.md
% for all four tiers; \nicefrac survives only inside ^{} and _{}, and as one of \Drv's styles.

%%%%%%%%%%%%%%%%%%%%%% Display-style array columns %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%
% `array` sets its cells in text style, so a \frac inside one comes out at script size. The
% `arr` filter writes the `>{\displaystyle}` prefixes itself for an equation hoisted into `eq:`;
% a solution file that writes \begin{array} by hand -- the monolingual trees, where hoisting
% buys nothing -- would otherwise have to repeat the prefix once per column. So:
%
%     \begin{array}{RCL}   instead of   \begin{array}{>{\displaystyle}r>{\displaystyle}c...}
%
% Upper case because l, c and r are taken, and mean text style -- still what a column of plain
% symbols wants.
\newcolumntype{L}{>{\displaystyle}l}
\newcolumntype{C}{>{\displaystyle}c}
\newcolumntype{R}{>{\displaystyle}r}
